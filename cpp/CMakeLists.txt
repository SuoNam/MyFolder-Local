cmake_minimum_required(VERSION 3.16)
project(myfolder_lan VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Drogon CONFIG REQUIRED)
if(TARGET SQLite3::SQLite3)
  set(MYFOLDER_SQLITE_TARGET SQLite3::SQLite3)
elseif(TARGET SQLite::SQLite3)
  set(MYFOLDER_SQLITE_TARGET SQLite::SQLite3)
else()
  set(MYFOLDER_SQLITE_TARGET ${SQLite3_LIBRARIES})
endif()
add_executable(myfolder-lan src/main.cpp)
target_include_directories(myfolder-lan PRIVATE third_party)
target_include_directories(myfolder-lan PRIVATE ${SQLite3_INCLUDE_DIRS})
target_link_libraries(myfolder-lan PRIVATE Drogon::Drogon OpenSSL::SSL OpenSSL::Crypto ${MYFOLDER_SQLITE_TARGET})
target_compile_definitions(myfolder-lan PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
if(WIN32)
  target_compile_options(myfolder-lan PRIVATE /utf-8 /W4)
  target_link_libraries(myfolder-lan PRIVATE ws2_32 crypt32)
else()
  target_compile_options(myfolder-lan PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(myfolder-lan PRIVATE pthread)
endif()
